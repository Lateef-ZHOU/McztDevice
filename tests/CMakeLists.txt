file(GLOB_RECURSE headers "${CMAKE_CURRENT_SOURCE_DIR}/../include/*.h")
add_executable(McztDevice_smoke smoke.cpp)
target_link_libraries(McztDevice_smoke PRIVATE Lidar::McztDevice LidarMcztDeviceplugin
    LidarTaskUIplugin LidarUICoreplugin LidarChartCoreplugin)
set(units)
foreach(header IN LISTS headers)
    file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}/../include" "${header}")
    string(MAKE_C_IDENTIFIER "${relative}" unit)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp" "#include <${relative}>\n")
    list(APPEND units "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp")
endforeach()
add_library(McztDevice_headers OBJECT ${units})
target_link_libraries(McztDevice_headers PRIVATE Lidar::McztDevice)
foreach(target McztDevice_headers McztDevice_smoke)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
add_test(NAME McztDevice_page COMMAND McztDevice_smoke)
set_tests_properties(McztDevice_page PROPERTIES TIMEOUT 30
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QSG_RHI_BACKEND=null"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
if(WIN32)
    get_filename_component(pcap_runtime_dir "${HANDLECORE_PCAP_RUNTIME}" DIRECTORY)
    set_property(TEST McztDevice_page APPEND PROPERTY ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:${pcap_runtime_dir}")
endif()
